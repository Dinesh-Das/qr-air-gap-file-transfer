import {spawn} from 'node:child_process';
import {createInterface} from 'node:readline';
import {readFile} from 'node:fs/promises';

/** Diagnostic-only: same Windows counters and strict missing-PID contract. */
export async function startResidentMemoryObserver(scriptPath){
 if(process.platform!=='win32')throw Error('Windows observer required');
 const original=await readFile(scriptPath,'utf8');
 const fence='$memoryProcessIds = [Console]::In.ReadToEnd() | ConvertFrom-Json';
 if(original.split(fence).length!==2)throw Error('Observer source mismatch');
 const source=`[Console]::Out.WriteLine('{"ready":true}')\nwhile ($null -ne ($memoryLine = [Console]::In.ReadLine())) {\n${original.replace(fence,'$memoryProcessIds = $memoryLine | ConvertFrom-Json')}\n}`;
 const child=spawn('powershell.exe',['-NoProfile','-NonInteractive','-Command',source],{windowsHide:true,stdio:['pipe','pipe','pipe']});
 let readyResolve,readyReject,pending=null,closed=false,stderr='',closing=false;
 const ready=new Promise((done,reject)=>{readyResolve=done;readyReject=reject;});
 const exited=new Promise(done=>child.once('close',(code,signal)=>{closed=true;const error=Error('Resident observer exited '+code+' '+signal+' '+stderr);readyReject(error);pending?.reject(error);pending=null;done({code,signal});}));
 child.once('error',e=>{readyReject(e);pending?.reject(e);pending=null;});
 child.stderr.setEncoding('utf8');child.stderr.on('data',s=>{stderr=(stderr+s).slice(-65536);});
 const lines=createInterface({input:child.stdout});
 lines.on('line',line=>{try{if(line.length>65536)throw Error('Oversized observer response');const value=JSON.parse(line.replace(/^\uFEFF/,''));if(value.ready===true){readyResolve();return;}if(!pending)throw Error('Unexpected observer response');const request=pending;pending=null;request.resolve(value);}catch(e){readyReject(e);pending?.reject(e);pending=null;child.kill();}});
 let startup;try{await Promise.race([ready,new Promise((_,reject)=>{startup=setTimeout(()=>{child.kill();reject(Error('Resident observer startup timeout'));},10000);})]);}finally{clearTimeout(startup);}
 return {pid:child.pid,source,
  async sample(ids){if(closed||closing)throw Error('Observer closed');if(!Array.isArray(ids)||!ids.length||ids.length>64||ids.some(id=>!Number.isSafeInteger(id)||id<1||id>2147483647))throw Error('Invalid process ID list.');if(pending)throw Error('Observer request already pending');let timer;try{return await new Promise((resolve,reject)=>{pending={resolve,reject};timer=setTimeout(()=>{pending=null;child.kill();reject(Error('Resident observer request timeout'));},10000);child.stdin.write(JSON.stringify([...new Set(ids)])+'\n',e=>{if(e){pending=null;reject(e);}});});}finally{clearTimeout(timer);}},
  async close(){closing=true;child.stdin.end();let timer;try{const result=await Promise.race([exited,new Promise((_,reject)=>{timer=setTimeout(()=>{child.kill();reject(Error('Resident observer did not close'));},5000);})]);lines.close();if(result.code!==0)throw Error('Observer close failed '+JSON.stringify(result));return result;}finally{clearTimeout(timer);}},
 };
}
