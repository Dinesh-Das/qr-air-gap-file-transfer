import {readFile,writeFile,mkdir,open} from 'node:fs/promises';
import {spawn} from 'node:child_process';
import {join,resolve} from 'node:path';

const runtime=resolve('artifacts/qrft-2026-10-16-device-names-portable/runtime/node.exe');
const folder=join(stage,'validation/unicode-fidelity-mutant-2026-10-16');await mkdir(folder);
const archive=await readFile('src/lib/archive.ts','utf8');if(!archive.includes('.normalize("NFC").toLowerCase()'))throw Error('Mutation target');
const mutated=archive.replaceAll('.normalize("NFC").toLowerCase()','.normalize("NFC").toUpperCase().toLowerCase()');
await writeFile(join(folder,'archive-fullfold-mutant.ts'),mutated,{flag:'wx'});
const test=await readFile('tests/unicode-path-fidelity.test.ts','utf8');
const mutantTest=test.replaceAll('"../src/','"../../src/').replace('"../../src/lib/archive"','"./archive-fullfold-mutant"');
await writeFile(join(folder,'unicode-mutant.test.ts'),mutantTest,{flag:'wx'});

const cli=join(stage,'node_modules/vitest/vitest.mjs');
for(const [name,args,expected] of [['fullfold-mutant',[cli,'run','--config',config],1],['type',[join(stage,'node_modules/typescript/bin/tsc'),'-b','--pretty','false'],0],['unit',[cli,'run'],0],['build',[join(stage,'node_modules/vite/bin/vite.js'),'build'],0]]){
 const path='release-evidence/unicode-fidelity-'+name+'-2026-10-16.log';const log=await open(path,'wx');let code;

 await writeFile('release-evidence/unicode-fidelity-'+name+'-exit-2026-10-16.txt',String(code)+'\n',{flag:'wx'});console.log(JSON.stringify({name,code,expected}));if(code!==expected)throw Error('Gate mismatch '+name);
 if(name==='fullfold-mutant'&&!/6 failed\s*\|\s*3 passed/.test(await readFile(path,'utf8')))throw Error('Mutation failed for unexpected reason');
}
