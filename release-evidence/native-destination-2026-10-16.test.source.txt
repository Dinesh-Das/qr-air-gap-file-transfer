// Diagnostic adapter only: Node filesystem operations, not a browser picker or
// File System Access implementation. Safe-write staging models close/abort.
function adapter(root){
 const writes=[];
 const checked=p=>{const rel=relative(root,resolve(p));if(rel==='..'||rel.startsWith('..\\')||isAbsolute(rel))throw Error('Outside owned fixture');return p;};
 const translate=e=>{if(e.code==='ENOENT')return new DOMException('Missing fixture entry','NotFoundError');if(['ENOTDIR','EISDIR'].includes(e.code))return new DOMException('Wrong fixture kind','TypeMismatchError');return e;};
 async function handle(p,kind){
  checked(p);
  let canonical;try{canonical=await realpath(p);const s=await lstat(p);if(s.isSymbolicLink())throw Error('Fixture links forbidden');if((kind==='directory')!==s.isDirectory())throw new DOMException('Wrong fixture kind','TypeMismatchError');}catch(e){throw translate(e);}
  checked(canonical);
  const h={name:basename(canonical),kind};
  if(kind==='directory')Object.assign(h,{
   async getDirectoryHandle(name,options={}){const child=checked(join(canonical,name));if(options.create)try{await mkdir(child);}catch(e){if(e.code!=='EEXIST')throw translate(e);}return handle(child,'directory');},
   async getFileHandle(name,options={}){const child=checked(join(canonical,name));if(options.create){let f;try{f=await open(child,'ax');}catch(e){if(e.code!=='EEXIST')throw translate(e);}finally{await f?.close();}}return handle(child,'file');},
   async *entries(){for(const entry of await readdir(canonical,{withFileTypes:true})){if(entry.isSymbolicLink())throw Error('Fixture links forbidden');yield [entry.name,await handle(join(canonical,entry.name),entry.isDirectory()?'directory':'file')];}},
   async removeEntry(name){await unlink(checked(join(canonical,name)));},
  });
  else Object.assign(h,{
   async getFile(){const data=await readFile(canonical);return new File([data],h.name);},
   async createWritable(){const staging=checked(join(dirname(canonical),'.qrft-adapter-'+randomUUID()+'.tmp'));const f=await open(staging,'wx');let done=false;
    return {async write(bytes){await f.writeFile(bytes);},async close(){await f.close();await rename(staging,canonical);done=true;writes.push(relative(root,canonical));},async abort(){if(!done){await f.close().catch(()=>{});await unlink(staging);done=true;}}};},
  });
  return h;
 }
 return {handle,writes};
}
function fixture(){
 const expected=new Map();let index=0;
 for(const pair of pairs)for(const name of pair){const path='Unicode/'+name+'.bin';expected.set(path,Buffer.from(`distinct native payload ${index++}: ${name}`));}
 expected.set('nested/文件/😀.bin',Buffer.from([0,255,17,128,37]));
 expected.set('zero.bin',Buffer.alloc(0));
 const ordered=[...expected].sort(([a],[b])=>a<b?-1:a>b?1:0);
 const bytes=Buffer.concat(ordered.map(([,b])=>b));
 const metadata=encodeLargeTreeMetadata({rootName:'received',directories:['Unicode','empty','nested','nested/文件'],files:ordered.map(([path,b])=>({path,size:b.length,lastModified:7,sha256Hex:digest(b)}))});
 const manifest=parseLargeTreeMetadata(metadata);
 const reader={size:BigInt(bytes.length),async read(offset,length){return new Uint8Array(bytes.subarray(Number(offset),Number(offset)+length));}};
 const binding={transferId:new Uint8Array(16).fill(19),streamSha256:new Uint8Array(createHash('sha256').update(bytes).digest())};
 return {manifest,reader,binding,expected:new Map(ordered),directories:['Unicode','empty','nested','nested/文件']};
}
async function inventory(root){
 const out=[];
 async function walk(p,prefix=''){for(const e of await readdir(p,{withFileTypes:true})){const path=prefix?prefix+'/'+e.name:e.name;if(e.isSymbolicLink())throw Error('Fixture links forbidden');if(e.isDirectory()){out.push({path,kind:'directory'});await walk(join(p,e.name),path);}else{const b=await readFile(join(p,e.name));out.push({path,kind:'file',bytes:b.length,sha256:digest(b)});}}}
 await walk(root);return out.sort((a,b)=>a.path<b.path?-1:a.path>b.path?1:0);
}
async function exact(root,f){const list=await inventory(root);expect(list.filter(e=>e.kind==='directory').map(e=>e.path)).toEqual(f.directories);expect(list.filter(e=>e.kind==='file').map(e=>e.path)).toEqual([...f.expected.keys()]);for(const [path,b] of f.expected)expect(await readFile(join(root,path))).toEqual(b);return list;}

 results.push({drive,root,cases,completeInventory,finalInventory:await inventory(root)});

Small synthetic bytes, one process, no Web Locks/browser/native picker/LAN/physical/power-loss/performance acceptance.',results},(_,v)=>typeof v==='bigint'?v.toString():v,2)+'\n');
